async function carregarVendasModernas(userId) { try { const container = document.getElementById('lista-vendas-moderna'); const contador = document.getElementById('contador-pedidos'); const { data, error } = await _supabase.from('vendas').select('*').eq('email_cliente', userId).order('id', { ascending: false }); if (error || !data || data.length === 0) { container.innerHTML = `

Nenhum pedido encontrado

`; contador.innerText = "0 pedidos"; lucide.createIcons(); return; } contador.innerText = `${data.length} ${data.length === 1 ? 'pedido' : 'pedidos'}`; container.innerHTML = data.map(venda => { let statusBadge = ''; let acaoHtml = ''; // AJUSTE: Badge muito pequeno (text-[10px]) e compacto (px-2 py-1) if (venda.status === 'Pago' || venda.status === 'Aprovado') { statusBadge = ` Pago`; acaoHtml = `Compra protegida por PagBank.`; } else { statusBadge = ` Aguardando`; acaoHtml = ``; } const dataFormatada = venda.created_at ? new Date(venda.created_at).toLocaleDateString('pt-BR') : '---'; return `

${venda.produto}

#${venda.id} | ${dataFormatada}

${statusBadge}

R$ ${venda.valor}

${acaoHtml}
`; }).join(''); lucide.createIcons(); } catch (e) { console.error("Erro vendas:", e); } }
